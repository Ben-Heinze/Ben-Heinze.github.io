Intro to binary logistic regression with the Donner Party data (Sleuth case 20.1): fitting =glm(..., family = "binomial")=, effects plots on the logit and probability scales, and drop-in-deviance tests.

#+begin_src R
options(show.signif.stars = FALSE)
library(Sleuth3)
library(tidyverse)
# rename dataframe 
donner_df <- case2001
glimpse(donner_df) 

# create new variables to match text
df_donner <- donner_df |> 
  mutate(fem = factor(ifelse(Sex == "Female", 1, 0)), 
         surv = factor(ifelse(Status == "Survived", 1, 0)))
# check new vars
glimpse(df_donner)

# fit binary logistic regression model 
fit_donner <- glm(surv ~ Age + fem, 
                  data = df_donner, family = "binomial") 
summary(fit_donner)
#+end_src

#+begin_src R
# fit the interaction (full) model
fit_dint <- glm(surv ~ fem*Age, data = df_donner, 
                family = "binomial")
# Wald-based tests are provided in summary output for 
# testing beta_j = 0 for just one beta. Which beta is assoc with the interaction? 
summary(fit_dint)
qnorm(p = 0.975)
#+end_src

#+begin_src R
# LRT is
anova(fit_donner, fit_dint)
confint(fit_dint)
exp(confint(fit_dint))
