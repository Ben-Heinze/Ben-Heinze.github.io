new_x <- data.frame(conc = seq(from = 49.06, to = 76.54, length.out = 100))

df_pred <- tibble(est = est_logit$fit, se = est_logit$se.fit) |>
  mutate(est_lower = est - 2*se,
         est_upper = est + 2*se,
         est_prob = exp(est)/(1 + exp(est)),
         prob_lower = exp(est_lower)/(1 + exp(est_lower)),

             alpha = 0.5, inherit.aes = FALSE) +

* Small habits worth keeping

- =|>= and =%>%= are interchangeable pipes. =pull()= turns a column into a vector, and =print(n = Inf)= shows every row of a tibble.
- Use =dplyr::select= when another loaded package (like =MASS=) also has a =select=.
- In a ggplot layer, =inherit.aes = FALSE= keeps it from picking up the plot's main aesthetics, which is useful when overlaying a second data frame.
- =pander::pander(summ_table, caption = "...")= gives a clean table in a knitted document.
- Use =round(summary(fit)$coef, 4)= to make the coefficient table readable.
