<script>
(function () {
  'use strict';
  var data = JSON.parse(document.getElementById('quiz-data').textContent);
  var root = document.getElementById('quiz');

  function norm(s) {
    return (s || '').trim().toLowerCase().replace(/\s+/g, ' ').replace(/[.’']/g, function (c) {
      return c === '.' ? '' : "'";
    }).replace(/[.]$/, '').trim();
  }

  var order = data.slice();      // current display order (shuffled in place)
  var nodes = {};                // n -> {el, feedback, get, set, item}

  function buildQuestion(item) {
    var q = document.createElement('div');
    q.className = 'quiz-q';
    q.id = 'q' + item.n;

    var prompt = document.createElement('div');
    prompt.className = 'quiz-prompt';
    prompt.innerHTML = '<span class="qnum">' + item.n + '.</span>';
    prompt.appendChild(document.createTextNode(item.q));
    q.appendChild(prompt);

    var getAnswer, setAnswer;

    if (item.type === 'mc' || item.type === 'tf') {
      var opts = item.type === 'tf' ? ['True', 'False'] : item.options;
      var ul = document.createElement('ul');
      ul.className = 'quiz-opts';
      opts.forEach(function (opt) {
        var li = document.createElement('li');
        var label = document.createElement('label');
        var radio = document.createElement('input');
        radio.type = 'radio';
        radio.name = 'q' + item.n;
        radio.value = opt;
        label.appendChild(radio);
        label.appendChild(document.createTextNode(' ' + opt));
        li.appendChild(label);
        ul.appendChild(li);
      });
      q.appendChild(ul);
      getAnswer = function () {
        var sel = q.querySelector('input[name="q' + item.n + '"]:checked');
        return sel ? sel.value : null;
      };
      setAnswer = function (val) {
        if (val == null) return;
        q.querySelectorAll('input[name="q' + item.n + '"]').forEach(function (r) {
          if (r.value === val) r.checked = true;
        });
      };
    } else if (item.type === 'fill') {
      var input = document.createElement('input');
      input.type = 'text';
      input.autocomplete = 'off';
      input.spellcheck = false;
      input.placeholder = 'Type your answer';
      q.appendChild(input);
      getAnswer = function () { return input.value; };
      setAnswer = function (val) { input.value = val || ''; };
    } else { // essay
      var ta = document.createElement('textarea');
      ta.placeholder = 'Write your answer, then Submit to see a model answer to self-assess.';
      q.appendChild(ta);
      getAnswer = function () { return ta.value; };
      setAnswer = function (val) { ta.value = val || ''; };
    }

    var fb = document.createElement('div');
    fb.className = 'quiz-feedback';
    q.appendChild(fb);

    nodes[item.n] = { el: q, feedback: fb, get: getAnswer, set: setAnswer, item: item };
    return q;
  }

  function renderAll() {
    root.innerHTML = '';
    nodes = {};
    order.forEach(function (item) { root.appendChild(buildQuestion(item)); });
  }

  renderAll();

  function clearScore() {
    var score = document.getElementById('quiz-score');
    score.className = 'quiz-score';
    score.innerHTML = '';
  }

  function shuffle() {
    // Preserve any answers already entered, then reorder and restore them.
    var saved = {};
    data.forEach(function (item) { if (nodes[item.n]) saved[item.n] = nodes[item.n].get(); });
    for (var i = order.length - 1; i > 0; i--) {
      var j = Math.floor(Math.random() * (i + 1));
      var t = order[i]; order[i] = order[j]; order[j] = t;
    }
    renderAll();
    data.forEach(function (item) { if (saved[item.n] != null) nodes[item.n].set(saved[item.n]); });
    clearScore();
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }

  function isCorrect(item, user) {
    if (user == null) return false;
    if (item.type === 'mc' || item.type === 'tf') return user === item.answer;
    if (item.type === 'fill') {
      var u = norm(user);
      return item.accept.some(function (a) { return norm(a) === u; });
    }
    return null; // essay: not auto-graded
  }

  function grade(showOnly) {
    var correct = 0, objective = 0;
    data.forEach(function (item) {
      var node = nodes[item.n];
      var user = node.get();
      var fb = node.feedback;
      node.el.classList.remove('is-correct', 'is-incorrect');
      fb.classList.remove('ok', 'bad', 'info');

      if (item.type === 'essay') {
        fb.classList.add('show', 'info');
        fb.innerHTML = '<strong>Model answer (self-assess):</strong> ' + escapeHtml(item.answer);
        return;
      }

      objective++;
      var ok = isCorrect(item, user);
      if (!showOnly && ok) correct++;
      fb.classList.add('show');

      if (showOnly) {
        fb.classList.add('info');
        fb.innerHTML = '<strong>Answer:</strong> <span class="quiz-c-ok">' +
          escapeHtml(item.answer) + '</span>' +
          (item.explain ? ' &mdash; ' + escapeHtml(item.explain) : '');
      } else if (ok) {
        node.el.classList.add('is-correct');
        fb.classList.add('ok');
        fb.innerHTML = '<span class="quiz-c-ok"><strong>Correct.</strong></span> ' +
          (item.explain ? escapeHtml(item.explain) : '');
      } else {
        node.el.classList.add('is-incorrect');
        fb.classList.add('bad');
        var yours = (user == null || user === '') ? '(blank)' : escapeHtml(user);
        fb.innerHTML = '<span class="quiz-c-bad"><strong>Incorrect.</strong></span> ' +
          'Your answer: ' + yours + '<br><strong>Correct:</strong> <span class="quiz-c-ok">' +
          escapeHtml(item.answer) + '</span>' +
          (item.explain ? ' &mdash; ' + escapeHtml(item.explain) : '');
      }
    });

    var score = document.getElementById('quiz-score');
    score.classList.add('show');
    if (showOnly) {
      score.innerHTML = 'Answers revealed. Objective questions: ' + objective +
        '. Essays (Q2, Q6, Q17): self-assess against the model answers.';
    } else {
      var pct = Math.round((correct / objective) * 100);
      score.innerHTML = 'Score: <span class="quiz-c-ok">' + correct + ' / ' + objective +
        '</span> objective correct (' + pct + '%). ' +
        'Essays (Q2, Q6, Q17) are self-assessed against the model answers below.';
    }
    score.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
  }

  function reset() {
    data.forEach(function (item) {
      var node = nodes[item.n];
      node.el.classList.remove('is-correct', 'is-incorrect');
      node.feedback.className = 'quiz-feedback';
      node.feedback.innerHTML = '';
      var inputs = node.el.querySelectorAll('input, textarea');
      inputs.forEach(function (i) {
        if (i.type === 'radio') i.checked = false; else i.value = '';
      });
    });
    clearScore();
    swReset();
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }

  function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }

  // ── Stopwatch (counts up when started; toggle pauses/resumes) ──────────────
  var swEl = document.getElementById('quiz-stopwatch');
  var swBtn = document.getElementById('quiz-timer-toggle');
  var swRunning = false, swStart = 0, swElapsed = 0, swInt = null;

  function fmt(ms) {
    var total = Math.floor(ms / 1000);
    var m = Math.floor(total / 60), s = total % 60;
    return (m < 10 ? '0' : '') + m + ':' + (s < 10 ? '0' : '') + s;
  }
  function swTick() {
    swEl.textContent = fmt(swElapsed + (swRunning ? Date.now() - swStart : 0));
  }
  function swStartFn() {
    if (swRunning) return;
    swRunning = true;
    swStart = Date.now();
    swEl.classList.add('running');
    swBtn.textContent = 'Pause';
    swInt = setInterval(swTick, 250);
    swTick();
  }
  function swPause() {
    if (!swRunning) return;
    swElapsed += Date.now() - swStart;
    swRunning = false;
    clearInterval(swInt);
    swEl.classList.remove('running');
    swBtn.textContent = 'Resume';
    swTick();
  }
  function swReset() {
    swRunning = false;
    clearInterval(swInt);
    swElapsed = 0;
    swEl.classList.remove('running');
    swBtn.textContent = 'Start timer';
    swEl.textContent = '00:00';
  }

  document.getElementById('quiz-submit').addEventListener('click', function () { grade(false); });
  document.getElementById('quiz-reveal').addEventListener('click', function () { grade(true); });
  document.getElementById('quiz-reset').addEventListener('click', reset);
  document.getElementById('quiz-shuffle').addEventListener('click', shuffle);
  swBtn.addEventListener('click', function () { swRunning ? swPause() : swStartFn(); });
}());
</script>
